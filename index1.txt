function clamp(
    value,
    min,
    max
) {

    return Math.max(
        min,
        Math.min(
            max,
            value
        )
    );

}


function easeOutCubic(t) {

    return 1 -
        Math.pow(
            1 - t,
            3
        );

}


function easeOutBack(t) {

    const c1 = 1.70158;
    const c3 = c1 + 1;

    return 1 +
        c3 *
        Math.pow(
            t - 1,
            3
        ) +
        c1 *
        Math.pow(
            t - 1,
            2
        );

}


/*
=========================================================
BACKGROUND
=========================================================
*/

function drawBackground(
    time
) {


    /*
    Base
    */

    const base =
        ctx.createLinearGradient(
            0,
            0,
            WIDTH,
            HEIGHT
        );


    base.addColorStop(
        0,
        "#020305"
    );

    base.addColorStop(
        .45,
        "#07131d"
    );

    base.addColorStop(
        1,
        "#010203"
    );


    ctx.fillStyle =
        base;

    ctx.fillRect(
        0,
        0,
        WIDTH,
        HEIGHT
    );


    /*
    Blue central glow
    */

    const pulse =
        .5 +
        Math.sin(
            time * 1.5
        ) * .1;


    const glow =
        ctx.createRadialGradient(
            WIDTH / 2,
            HEIGHT * .48,
            0,
            WIDTH / 2,
            HEIGHT * .48,
            500
        );


    glow.addColorStop(
        0,
        `rgba(10,110,180,${.20 * pulse})`
    );

    glow.addColorStop(
        .45,
        "rgba(10,70,120,.07)"
    );

    glow.addColorStop(
        1,
        "rgba(0,0,0,0)"
    );


    ctx.fillStyle =
        glow;

    ctx.fillRect(
        0,
        0,
        WIDTH,
        HEIGHT
    );


    /*
    Gold bottom glow
    */

    const gold =
        ctx.createRadialGradient(
            WIDTH / 2,
            HEIGHT,
            0,
            WIDTH / 2,
            HEIGHT,
            500
        );


    gold.addColorStop(
        0,
        "rgba(255,190,50,.11)"
    );

    gold.addColorStop(
        1,
        "rgba(255,190,50,0)"
    );


    ctx.fillStyle =
        gold;

    ctx.fillRect(
        0,
        0,
        WIDTH,
        HEIGHT
    );


    /*
    Diagonal stadium lines
    */

    ctx.save();

    ctx.translate(
        WIDTH / 2,
        HEIGHT / 2
    );

    ctx.rotate(
        -8 * Math.PI / 180
    );


    for (
        let i = -5;
        i <= 5;
        i++
    ) {

        const y =
            i * 130;


        ctx.strokeStyle =
            "rgba(255,210,80,.06)";

        ctx.lineWidth = 2;


        ctx.beginPath();

        ctx.moveTo(
            -WIDTH,
            y
        );

        ctx.lineTo(
            WIDTH,
            y
        );

        ctx.stroke();

    }

    ctx.restore();


    /*
    Moving light sweep
    */

    const sweepProgress =
        (time / 6) % 1;


    const sweepX =
        -500 +
        sweepProgress *
        (WIDTH + 1000);


    ctx.save();

    ctx.translate(
        sweepX,
        0
    );

    ctx.rotate(
        20 * Math.PI / 180
    );


    const sweep =
        ctx.createLinearGradient(
            -120,
            0,
            120,
            0
        );


    sweep.addColorStop(
        0,
        "rgba(255,210,80,0)"
    );

    sweep.addColorStop(
        .5,
        "rgba(255,230,150,.13)"
    );

    sweep.addColorStop(
        1,
        "rgba(255,210,80,0)"
    );


    ctx.fillStyle =
        sweep;

    ctx.fillRect(
        -120,
        -HEIGHT,
        240,
        HEIGHT * 2
    );


    ctx.restore();

}


/*
=========================================================
PARTICLES
=========================================================
*/

const particles = [];

for (
    let i = 0;
    i < 35;
    i++
) {

    particles.push({

        x:
            Math.random() *
            WIDTH,

        y:
            Math.random() *
            HEIGHT,

        size:
            1 +
            Math.random() * 3,

        speed:
            .5 +
            Math.random() * 1.5,

        phase:
            Math.random() * 10

    });

}


function drawParticles(
    time
) {

    for (
        const p of particles
    ) {


        const y =
            p.y -
            (
                time *
                20 *
                p.speed
            ) %
            HEIGHT;


        const alpha =
            .15 +
            .45 *
            (
                .5 +
                .5 *
                Math.sin(
                    time * 2 +
                    p.phase
                )
            );


        ctx.fillStyle =
            `rgba(255,215,100,${alpha})`;


        ctx.beginPath();

        ctx.arc(
            p.x,
            y < 0
                ? y + HEIGHT
                : y,
            p.size,
            0,
            Math.PI * 2
        );

        ctx.fill();

    }

}

/*
=========================================================
MAIN TITLE
=========================================================
*/

function drawTitle(
    time
) {


    const center =
        WIDTH / 2;


    /*
    Title starts
    */

    const startY =
        HEIGHT * .39;


    /*
    -----------------------------------------------
    FA
    -----------------------------------------------
    */

    let faProgress =
        clamp(
            (time - .35) / .6,
            0,
            1
        );


    faProgress =
        easeOutCubic(
            faProgress
        );


    ctx.save();


    ctx.globalAlpha =
        faProgress;


    ctx.translate(
        0,
        (1 - faProgress) *
        45
    );


    ctx.shadowColor =
        "rgba(0,0,0,.8)";

    ctx.shadowBlur =
        25;


    drawCenteredText(

        "FA",

        center,

        startY,

        "900 82px Arial",

        "#ffffff"

    );


    ctx.restore();


    /*
    -----------------------------------------------
    COMMUNITY
    -----------------------------------------------
    */

    let communityProgress =
        clamp(
            (time - .65) / .65,
            0,
            1
        );


    communityProgress =
        easeOutCubic(
            communityProgress
        );


    ctx.save();


    ctx.globalAlpha =
        communityProgress;


    ctx.translate(
        0,
        (1 - communityProgress) *
        40
    );


    ctx.shadowColor =
        "rgba(0,0,0,.8)";

    ctx.shadowBlur =
        20;


    drawCenteredText(

        "COMMUNITY",

        center,

        startY + 92,

        "900 45px Arial",

        "#ffffff"

    );


    ctx.restore();


    /*
    -----------------------------------------------
    SHIELD TYPING
    -----------------------------------------------
    */


    const word =
        "SHIELD";


    const typingStart =
        1.15;


    const letterDelay =
        .30;


    const letters = [];


    for (
        let i = 0;
        i < word.length;
        i++
    ) {


        const appearTime =
            typingStart +
            i *
            letterDelay;


        const progress =
            clamp(
                (
                    time -
                    appearTime
                ) / .22,
                0,
                1
            );


        letters.push({

            char:
                word[i],

            progress:
                easeOutBack(
                    progress
                ),

            visible:
                time >=
                appearTime

        });

    }


    /*
    Measure text
    */

    ctx.font =
        "900 74px Arial";


    const letterWidth =
        ctx.measureText("M").width;


    const totalWidth =
        word.length *
        letterWidth;


    let x =
        center -
        totalWidth / 2;


    const shieldY =
        startY + 180;


    /*
    Gold gradient
    */

    const gold =
        ctx.createLinearGradient(
            0,
            shieldY - 50,
            0,
            shieldY + 50
        );


    gold.addColorStop(
        0,
        "#fff3a3"
    );

    gold.addColorStop(
        .35,
        "#ffd85c"
    );

    gold.addColorStop(
        1,
        "#d59d20"
    );


    /*
    Draw each letter
    */

    for (
        let i = 0;
        i < letters.length;
        i++
    ) {


        const l =
            letters[i];


        if (
            !l.visible
        ) {

            continue;

        }


        ctx.save();


        ctx.globalAlpha =
            clamp(
                (
                    time -
                    (
                        typingStart +
                        i *
                        letterDelay
                    )
                ) / .18,
                0,
                1
            );


        const scale =
            .9 +
            .1 *
            l.progress;


        ctx.translate(
            x +
            letterWidth / 2,
            shieldY
        );


        ctx.scale(
            scale,
            scale
        );


        ctx.shadowColor =
            "rgba(255,205,70,.35)";

        ctx.shadowBlur =
            25;


        ctx.font =
            "900 74px Arial";

        ctx.textAlign =
            "center";

        ctx.textBaseline =
            "middle";

        ctx.fillStyle =
            gold;


        ctx.fillText(
            l.char,
            0,
            0
        );


        ctx.restore();


        x +=
            letterWidth;

    }


    /*
    -----------------------------------------------
    CURSOR
    -----------------------------------------------
    */


    const lastTyped =
        Math.min(
            Math.floor(
                (
                    time -
                    typingStart
                ) /
                letterDelay
            ) + 1,
            word.length
        );


    if (
        lastTyped > 0
    ) {


        const typedWidth =
            lastTyped *
            letterWidth;


        const cursorX =
            center -
            totalWidth / 2 +
            typedWidth +
            6;


        /*
        Cursor starts appearing
        immediately after first letter.
        */


        const cursorBlink =
            Math.sin(
                time *
                Math.PI *
                3
            ) > 0;


        /*
        Keep cursor visible
        */

        ctx.fillStyle =
            cursorBlink
                ? "#ffd45c"
                : "rgba(255,212,92,.08)";


        ctx.shadowColor =
            "#ffd45c";

        ctx.shadowBlur =
            15;


        ctx.fillRect(

            cursorX,

            shieldY - 35,

            4,

            70

        );

    }

}

/*
=========================================================
ROUNDED RECT
=========================================================
*/

function roundRect(
    context,
    x,
    y,
    width,
    height,
    radius
) {


    context.beginPath();

    context.moveTo(
        x + radius,
        y
    );

    context.lineTo(
        x + width - radius,
        y
    );

    context.quadraticCurveTo(
        x + width,
        y,
        x + width,
        y + radius
    );

    context.lineTo(
        x + width,
        y + height - radius
    );

    context.quadraticCurveTo(
        x + width,
        y + height,
        x + width - radius,
        y + height
    );

    context.lineTo(
        x + radius,
        y + height
    );

    context.quadraticCurveTo(
        x,
        y + height,
        x,
        y + height - radius
    );

    context.lineTo(
        x,
        y + radius
    );

    context.quadraticCurveTo(
        x,
        y,
        x + radius,
        y
    );

    context.closePath();

}

/*
=========================================================
ANIMATION LOOP
=========================================================
*/

let animationID = null;


function startPreview() {


    const start =
        performance.now();


    function loop(
        now
    ) {


        const time =
            (
                now -
                start
            ) / 1000;


        drawFrame(
            time
        );


        if (
            time <
            DURATION
        ) {

            animationID =
                requestAnimationFrame(
                    loop
                );

        }
        else {

            drawFrame(
                DURATION
            );

        }

    }


    cancelAnimationFrame(
        animationID
    );


    requestAnimationFrame(
        loop
    );

}

let ffmpeg = null;

let ffmpegLoaded = false;


/*
---------------------------------------------------------
LOAD FFMPEG
---------------------------------------------------------
*/

async function loadFFmpeg() {


    if (
        ffmpegLoaded
    ) {

        return;

    }


    status.textContent =
        "Loading MP4 engine...";


    /*
    Global objects supplied by
    the UMD builds.
    */

    const FFmpegClass =
        FFmpegWASM.FFmpeg;


    const {
        fetchFile,
        toBlobURL
    } =
        FFmpegUtil;


    ffmpeg =
        new FFmpegClass();


    /*
    Optional log
    */

    ffmpeg.on(
        "log",
        ({
            message
        }) => {

            console.log(
                "[FFmpeg]",
                message
            );

        }
    );


    /*
    Official single-thread core.

    We deliberately use the
    single-thread version so that
    GitHub Pages does not need
    SharedArrayBuffer headers.
    */

    const baseURL =
        "https://cdn.jsdelivr.net/npm/@ffmpeg/core@0.12.10/dist/umd";


    await ffmpeg.load({

        coreURL:
            await toBlobURL(

                `${baseURL}/ffmpeg-core.js`,

                "text/javascript"

            ),

        wasmURL:
            await toBlobURL(

                `${baseURL}/ffmpeg-core.wasm`,

                "application/wasm"

            )

    });


    ffmpegLoaded =
        true;


    console.log(
        "FFmpeg loaded"
    );

}


/*
=========================================================
RECORD CANVAS
=========================================================
*/

async function recordVideo() {


    /*
    Check MediaRecorder
    */

    if (
        !window.MediaRecorder
    ) {

        throw new Error(
            "MediaRecorder is not supported by this browser."
        );

    }


    /*
    Create canvas stream
    */

    const stream =
        canvas.captureStream(
            FPS
        );


    /*
    Try VP9
    */

    let mime =
        "video/webm;codecs=vp9";


    /*
    Safari/iOS may not support
    VP9, so fallback.
    */

    if (
        !MediaRecorder.isTypeSupported(
            mime
        )
    ) {

        mime =
            "video/webm;codecs=vp8";

    }


    if (
        !MediaRecorder.isTypeSupported(
            mime
        )
    ) {

        mime =
            "video/webm";

    }


    const recorder =
        new MediaRecorder(

            stream,

            {

                mimeType:
                    mime,

                videoBitsPerSecond:
                    8_000_000

            }

        );


    const chunks = [];


    recorder.ondataavailable =
        event => {

            if (
                event.data &&
                event.data.size
            ) {

                chunks.push(
                    event.data
                );

            }

        };


    /*
    Start
    */

    recorder.start();


    /*
    Render exact duration
    */

    const totalFrames =
        DURATION *
        FPS;


    const start =
        performance.now();


    for (
        let frame = 0;
        frame <= totalFrames;
        frame++
    ) {


        /*
        Use deterministic
        frame time.

        This means the typing
        animation is identical
        every export.
        */

        const time =
            frame /
            FPS;


        drawFrame(
            time
        );


        /*
        Update UI
        */

        const percent =
            (
                frame /
                totalFrames
            ) * 100;


        progressBar.style.width =
            `${percent}%`;


        status.textContent =
            `Rendering ${Math.round(percent)}%`;


        /*
        Give browser a chance
        to actually capture frame.
        */

        await new Promise(
            resolve =>
                requestAnimationFrame(
                    resolve
                )
        );

    }


    /*
    Stop recorder
    */

    recorder.stop();


    await new Promise(
        resolve => {

            recorder.onstop =
                resolve;

        }
    );


    /*
    Stop tracks
    */

    stream
        .getTracks()
        .forEach(
            track =>
                track.stop()
        );


    return new Blob(
        chunks,
        {
            type: mime
        }
    );

}


/*
=========================================================
CONVERT WEBM → MP4
=========================================================
*/

async function convertToMP4(
    webmBlob
) {


    await loadFFmpeg();


    const {
        fetchFile
    } =
        FFmpegUtil;


    status.textContent =
        "Preparing MP4...";


    /*
    Write WebM
    */

    await ffmpeg.writeFile(

        "input.webm",

        await fetchFile(
            webmBlob
        )

    );


    /*
    Encode H.264 MP4
    */

    await ffmpeg.exec([

        "-i",
        "input.webm",

        "-c:v",
        "libx264",

        "-preset",
        "ultrafast",

        "-crf",
        "23",

        "-pix_fmt",
        "yuv420p",

        "-r",
        String(FPS),

        "-vf",
        `scale=${WIDTH}:${HEIGHT}`,

        "-movflags",
        "+faststart",

        "FA_Community_Shield.mp4"

    ]);


    /*
    Read MP4
    */

    const data =
        await ffmpeg.readFile(
            "FA_Community_Shield.mp4"
        );


    /*
    Cleanup
    */

    try {

        await ffmpeg.deleteFile(
            "input.webm"
        );

        await ffmpeg.deleteFile(
            "FA_Community_Shield.mp4"
        );

    }

    catch (e) {

        console.log(
            "Cleanup warning:",
            e
        );

    }


    return new Blob(

        [data.buffer],

        {
            type:
                "video/mp4"
        }

    );

}


/*
=========================================================
DOWNLOAD
=========================================================
*/

function downloadMP4(
    blob
) {


    const url =
        URL.createObjectURL(
            blob
        );


    const link =
        document.createElement(
            "a"
        );


    link.href =
        url;


    link.download =
        "FA_Community_Shield.mp4";


    document.body.appendChild(
        link
    );


    link.click();


    link.remove();


    /*
    Keep URL alive briefly
    for mobile browsers.
    */

    setTimeout(
        () => {

            URL.revokeObjectURL(
                url
            );

        },
        10000
    );

}
