<script>

  /*
   * GANTI URL INI
   * dengan URL endpoint Worker milikmu.
   */
  const WORKER_URL =
    "https://dry-band-fc82.sabar-41c.workers.dev/data";


  let running = false;
  let controller = null;

  let totalBytes = 0;

  let previousBytes = 0;
  let previousTime = performance.now();


  const usageElement =
    document.getElementById("usage");

  const speedElement =
    document.getElementById("speed");

  const statusElement =
    document.getElementById("status");

  const startButton =
    document.getElementById("start");

  const stopButton =
    document.getElementById("stop");


  function formatMB(bytes) {

    return (
      bytes / 1024 / 1024
    ).toFixed(2) + " MB";

  }


  function updateDisplay() {

    usageElement.textContent =
      formatMB(totalBytes);


    const now =
      performance.now();


    const elapsed =
      (now - previousTime) / 1000;


    /*
     * Update speed setiap 500 ms.
     * Ini membuat angka speed tidak berkedip
     * terlalu cepat.
     */
    if (elapsed >= 0.5) {

      const bytes =
        totalBytes - previousBytes;


      const mbps =
        bytes /
        1024 /
        1024 /
        elapsed;


      speedElement.textContent =
        mbps.toFixed(2) + " MB/s";


      previousBytes =
        totalBytes;

      previousTime =
        now;

    }

  }


  async function downloadStream() {

    controller =
      new AbortController();


    try {

      const response =
        await fetch(
          WORKER_URL +
          "?t=" +
          Date.now(),
          {
            method: "GET",

            cache: "no-store",

            signal:
              controller.signal
          }
        );


      if (!response.ok) {

        throw new Error(
          "HTTP " +
          response.status
        );

      }


      if (!response.body) {

        throw new Error(
          "Browser tidak mendukung streaming."
        );

      }


      const reader =
        response.body.getReader();


      try {

        /*
         * Loop membaca stream.
         */
        while (running) {

          const {
            value,
            done
          } =
            await reader.read();


          /*
           * Stream selesai.
           */
          if (done) {
            break;
          }


          /*
           * Jangan membuat Blob.
           * Jangan membuat file.
           *
           * Kita hanya menghitung ukuran
           * chunk yang diterima.
           */
          if (value) {

            totalBytes +=
              value.byteLength;


            updateDisplay();

          }

        }

      } finally {

        /*
         * Lepaskan reader ketika selesai
         * atau ketika STOP ditekan.
         */
        try {

          await reader.cancel();

        } catch (_) {}

      }

    } finally {

      controller = null;

    }

  }


  async function start() {

    if (running) {
      return;
    }


    running = true;


    totalBytes = 0;

    previousBytes = 0;

    previousTime =
      performance.now();


    usageElement.textContent =
      "0.00 MB";


    speedElement.textContent =
      "0.00 MB/s";


    startButton.disabled =
      true;

    stopButton.disabled =
      false;


    statusElement.textContent =
      "Berjalan...";


    /*
     * LOOP UTAMA
     *
     * Setiap request menunggu sampai
     * stream sebelumnya selesai.
     *
     * Jika STOP ditekan, running menjadi
     * false dan loop berhenti.
     */
    while (running) {

      try {

        await downloadStream();

      } catch (error) {

        /*
         * Abort dari tombol STOP bukan error
         * yang perlu ditampilkan.
         */
        if (
          error.name ===
          "AbortError"
        ) {

          break;

        }


        statusElement.textContent =
          "Error: " +
          error.message;


        running = false;

      }

    }


    startButton.disabled =
      false;

    stopButton.disabled =
      true;


    statusElement.textContent =
      "Berhenti — Total: " +
      formatMB(totalBytes);

  }


  function stop() {

    if (!running) {
      return;
    }


    /*
     * Matikan loop utama.
     */
    running = false;


    /*
     * Batalkan request yang sedang berjalan.
     */
    if (controller) {

      controller.abort();

    }


    statusElement.textContent =
      "Menghentikan...";

  }


  startButton.addEventListener(
    "click",
    start
  );


  stopButton.addEventListener(
    "click",
    stop
  );

</script>
